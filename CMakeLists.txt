cmake_minimum_required(VERSION 3.20)
project(volterra_options VERSION 0.1.0 LANGUAGES CXX)

option(VOLTERRA_BUILD_EXAMPLES "Build command-line examples" ON)
option(VOLTERRA_BUILD_TESTS "Build numerical validation tests" ON)
option(VOLTERRA_ENABLE_NATIVE "Enable host-specific compiler optimizations" OFF)

add_library(volterra src/options.cpp)
add_library(volterra::volterra ALIAS volterra)
target_include_directories(volterra PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(volterra PUBLIC cxx_std_23)
set_target_properties(volterra PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(volterra PRIVATE /W4 /permissive-)
else()
  target_compile_options(volterra PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(VOLTERRA_ENABLE_NATIVE)
    target_compile_options(volterra PRIVATE -march=native)
  endif()
endif()
if(VOLTERRA_BUILD_EXAMPLES)
  add_executable(volterra_demo examples/pricing_demo.cpp)
  target_link_libraries(volterra_demo PRIVATE volterra::volterra)
endif()

if(VOLTERRA_BUILD_TESTS)
  enable_testing()
  add_executable(volterra_tests tests/options_tests.cpp)
  target_link_libraries(volterra_tests PRIVATE volterra::volterra)
  add_test(NAME volterra_numerics COMMAND volterra_tests)
endif()
